<script id="demo-page-script">
const ENDPOINT = "/_api/serverlogics/customer-summary";

function errorText(payload) {
  var text = payload && payload.message != null ? String(payload.message) : "";
  if (payload && Array.isArray(payload.errors)) {
    var extra = payload.errors
      .map((item) => (item && item.message ? String(item.message) : ""))
      .filter((item) => item !== "")
      .join(" ");
    if (extra) text = text ? text + " " + extra : extra;
  }
  return text;
}

function showResult(text, isError) {
  var result = document.getElementById("result");
  result.textContent = text;
  result.className = isError ? "sl-result sl-error" : "sl-result";
}

function showEnvelope(resp) {
  if (!resp || typeof resp !== "object" || Array.isArray(resp)) {
    showResult("The response could not be read.", true);
    return;
  }
  if (resp.success === false) {
    showResult(
      resp.error == null || resp.error === "" ? "Server logic not available" : String(resp.error),
      true,
    );
    return;
  }
  if (resp.success !== true || typeof resp.data !== "string") {
    showResult("The response could not be read.", true);
    return;
  }
  var payload;
  try {
    payload = JSON.parse(resp.data);
  } catch (e) {
    showResult("The response could not be read.", true);
    return;
  }
  if (!payload || typeof payload !== "object" || Array.isArray(payload)) {
    showResult("The response could not be read.", true);
    return;
  }
  if (payload.status === "success") {
    showResult(payload.message == null ? "" : String(payload.message), false);
    return;
  }
  showResult(errorText(payload) || "Application response failed.", true);
}

function callServer() {
  var nameInput = document.getElementById("nameInput");
  var name = nameInput.value || "World";
  showResult("Loading...", false);
  shell.getTokenDeferred().done((token) => {
    fetch(ENDPOINT + "?name=" + encodeURIComponent(name), {
      method: "GET",
      headers: { "__RequestVerificationToken": token },
    })
      .then((res) => res.json())
      .then((resp) => {
        showEnvelope(resp);
      })
      .catch(() => {
        showResult("Request failed", true);
      });
  }).fail(() => {
    showResult("Request failed", true);
  });
}

document.getElementById("callBtn").onclick = callServer;
</script>
