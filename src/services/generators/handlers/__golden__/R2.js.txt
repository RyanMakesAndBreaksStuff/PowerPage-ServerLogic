// R2: Validate a request

function post() {
  let body;
  try {
    body = JSON.parse(Server.Context.Body);
  } catch {
    const errors = [{ field: "body", message: "Body must be a JSON object." }];
    return JSON.stringify({ status: "error", message: "The request isn't valid.", errors: errors });
  }
  if (body === null || Array.isArray(body) || typeof body !== "object") {
    const errors = [{ field: "body", message: "Body must be a JSON object." }];
    return JSON.stringify({ status: "error", message: "The request isn't valid.", errors: errors });
  }
  const name = typeof body.name === "string" ? body.name.trim() : body.name;
  const email = typeof body.email === "string" ? body.email.trim() : body.email;
  const errors = [];
  if (typeof name !== "string" || name.length === 0 || name.length > 100) {
    errors.push({ field: "name", message: "Name is required and must be 100 characters or fewer." });
  }
  let emailOk = false;
  if (typeof email === "string" && email.length <= 254) {
    const parts = email.split("@");
    emailOk = parts.length === 2 && parts[1].indexOf(".") !== -1;
  }
  if (!emailOk) {
    errors.push({ field: "email", message: "Email must be a valid address of 254 characters or fewer." });
  }
  if (errors.length > 0) {
    Server.Logger.Warn("R2 post rejected");
    return JSON.stringify({ status: "error", message: "The request isn't valid.", errors: errors });
  }
  return JSON.stringify({ status: "success", name, email });
}
