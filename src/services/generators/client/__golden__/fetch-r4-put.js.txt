// dataverse-crud | PUT | put
const params = new URLSearchParams({
  "entitySetName": "contacts",
  "id": "00000000-0000-0000-0000-000000000000",
});
const url = "/_api/serverlogics/dataverse-crud?" + params;

function showError(text) {
  console.error(text);
}

function handleEnvelope(envelope) {
  if (envelope.success === false) {
    showError(String(envelope.error));
    return;
  }
  let parsed;
  try {
    parsed = JSON.parse(envelope.data);
  } catch {
    showError("The response data could not be read.");
    return;
  }
  if (parsed.status !== "success") {
    showError(String(parsed.message));
    return;
  }
  console.log("done");
}

shell.getTokenDeferred().done(function (token) {
  fetch(url, {
    method: "PUT",
    headers: {
      "__RequestVerificationToken": token,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(
      {
        "telephone1": "555-123-4567"
      }
    ),
  })
    .then(function (response) {
      return response.json();
    })
    .then(handleEnvelope)
    .catch(function () {
      showError("The call failed before a response envelope could be read.");
    });
}).fail(function () {
  showError("The request token was rejected.");
});
